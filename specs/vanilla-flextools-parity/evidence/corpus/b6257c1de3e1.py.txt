#
#   doubleref
#    - A FlexTools Module -
#
#   List entries that are registered as both a variant and a complex form
#   of the same target entry (a redundant / double relationship), including
#   the variant type and complex form type involved.
#
#   API Target: flexicon
#   Platforms: Python .NET and IronPython

from flextoolslib import *
from flexicon import FLExProject, LexEntryOperations, VariantOperations
from SIL.LCModel import ILexEntry

#----------------------------------------------------------------
# Documentation that the user sees:

docs = {FTM_Name        : "doubleref",
        FTM_Version     : 1,
        FTM_ModifiesDB  : False,
        FTM_Synopsis    : "List entries that are both a variant and a complex form of the same entry",
        FTM_Description :
"""
Finds and lists entries that have a "variant of" relationship AND a "complex form"
relationship pointing at the same target entry, and lists them along with the
variant type and complex form type of each relationship.
""" }

#----------------------------------------------------------------
# The main processing function

def Main(project, report, modifyAllowed):
    """
    Main entry point for the FlexTools module.

    Args:
        project: FLExProject instance providing access to the FieldWorks database
        report: Reporter object for logging (report.Info, report.Warning, report.Error)
        modifyAllowed: Boolean indicating if database modifications are permitted
    """
    lex = LexEntryOperations(project)
    variants = VariantOperations(project)

    report.Info("Scanning for entries that are both a variant and a complex form of the same entry...")

    for entry in project.LexEntry.GetAll():
        lex_entry = ILexEntry(entry)
        headword = project.LexEntry.GetHeadword(lex_entry)

        variant_targets = []
        variant_type_names = []
        for v in variants.GetAll(entry):
            v_type = variants.GetType(v)
            type_name = variants.GetTypeName(v_type) if v_type else "(unspecified variant type)"
            for comp in v.ComponentLexemesRS:
                variant_targets.append(ILexEntry(comp))
                variant_type_names.append(type_name)

        complex_targets = []
        complex_type_names = []
        for ref in lex_entry.ComplexFormEntryRefs:
            cf_type = project.LexiconGetComplexFormType(ref)
            type_name = cf_type.Name.BestAnalysisAlternative.Text if cf_type else "(unspecified complex form type)"
            for comp in ref.ComponentLexemesRS:
                complex_targets.append(comp)
                complex_type_names.append(type_name)

        for i, target in enumerate(variant_targets):
            if target in complex_targets:
                j = complex_targets.index(target)
                target_headword = project.LexEntry.GetHeadword(target)
                report.Info(
                    f"Entry '{headword}' is both a variant (type: {variant_type_names[i]}) "
                    f"and a complex form (type: {complex_type_names[j]}) of: {target_headword}"
                )

    report.Info("Done.")


#----------------------------------------------------------------

FlexToolsModule = FlexToolsModuleClass(Main, docs)

#----------------------------------------------------------------
if __name__ == '__main__':
    print(FlexToolsModule.Help())
